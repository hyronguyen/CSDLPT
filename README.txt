CREATE PROCEDURE GetProductsAbovePrice (@Price INT)
AS
BEGIN
    -- Select products from SANPHAM_MANH1 and SANPHAM_MANH2 (combining the information from these two tables)
    SELECT 
        SANPHAM_MANH1.MASANPHAM, 
        SANPHAM_MANH1.TENSANPHAM, 
        SANPHAM_MANH2.GIABAN,
        SANPHAM_MANH2.MAKHOHANG
    FROM 
        SANPHAM_MANH1
    INNER JOIN 
        SANPHAM_MANH2 
    ON 
        SANPHAM_MANH1.MASANPHAM = SANPHAM_MANH2.MASANPHAM
    WHERE 
        SANPHAM_MANH2.GIABAN > @Price

    UNION

    -- Select products from SANPHAM_MANH3
    SELECT 
        SANPHAM_MANH3.MASANPHAM, 
        SANPHAM_MANH3.TENSANPHAM, 
        SANPHAM_MANH3.GIABAN,
        SANPHAM_MANH3.MAKHOHANG
    FROM 
        SANPHAM_MANH3
    WHERE 
        SANPHAM_MANH3.GIABAN > @Price
END
GO

CREATE PROCEDURE GetProductsInRange (@MinPrice INT, @MaxPrice INT)
AS
BEGIN
    -- Select products from SANPHAM_MANH1 and SANPHAM_MANH2 (combining the information from these two tables)
    SELECT 
        SANPHAM_MANH1.MASANPHAM, 
        SANPHAM_MANH1.TENSANPHAM, 
        SANPHAM_MANH2.GIABAN,
        SANPHAM_MANH2.MAKHOHANG
    FROM 
        SANPHAM_MANH1
    INNER JOIN 
        SANPHAM_MANH2 
    ON 
        SANPHAM_MANH1.MASANPHAM = SANPHAM_MANH2.MASANPHAM
    WHERE 
        SANPHAM_MANH2.GIABAN BETWEEN @MinPrice AND @MaxPrice

    UNION

    -- Select products from SANPHAM_MANH3
    SELECT 
        SANPHAM_MANH3.MASANPHAM, 
        SANPHAM_MANH3.TENSANPHAM, 
        SANPHAM_MANH3.GIABAN,
        SANPHAM_MANH3.MAKHOHANG
    FROM 
        SANPHAM_MANH3
    WHERE 
        SANPHAM_MANH3.GIABAN BETWEEN @MinPrice AND @MaxPrice
END
GO
